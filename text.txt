local BRANCH = "main"
local ROOT = "https://raw.githubusercontent.com/"..GITHUB_USER.."/"..GITHUB_REPO.."/"..BRANCH.."/"

local Players = game:GetService("Players")
local RunService = game:GetService("RunService")
local LocalPlayer = Players.LocalPlayer

-- 1. Cargamos TU librería visual (Chibi/Kawaii)
local UI_Library = loadstring(game:HttpGet(ROOT .. "design.lua?v=" .. os.time()))()
local Window = UI_Library.CreateWindow({ 
    Title = "FROGGY HUB", 
    Subtitle = "SURVIVE YOUR IDEAS" 
})
local TabMain = Window:CreateTab("PRINCIPAL")

local GodModeEnabled = false
local healthLoop = nil

-- 🔥 Hook para bloquear TakeDamage (Evita que la lava/nuke te resten vida)
local function hookHumanoid(hum)
    if not hum then return end
    if hookfunction then
        pcall(function()
            local oldTakeDamage = hookfunction(hum.TakeDamage, function(self, amount)
                if GodModeEnabled then return end
                return oldTakeDamage(self, amount)
            end)
        end)
    end
end

-- Función principal del GodMode
local function setupGodMode(char)
    if not char then return end
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hum then return end

    -- Subimos la vida, pero de forma segura
    hum.MaxHealth = 5000
    hum.Health = 5000

    hookHumanoid(hum)

    if healthLoop then healthLoop:Disconnect() end

    -- Bucle de auto-curación (Si la lava te quita vida, te la devuelve al instante)
    healthLoop = RunService.Heartbeat:Connect(function()
        if GodModeEnabled and hum.Parent then
            if hum.Health < hum.MaxHealth then
                hum.Health = hum.MaxHealth
            end
            -- Si el tornado te tira, te levanta
            if hum.PlatformStand then
                hum.PlatformStand = false
            end
        end
    end)
end

if LocalPlayer.Character then setupGodMode(LocalPlayer.Character) end
LocalPlayer.CharacterAdded:Connect(function(char)
    task.wait(1)
    setupGodMode(char)
end)

-- 2. Interfaz visual
TabMain:AddSection("SUPERVIVENCIA")

TabMain:AddToggle("MODO DIOS (INMORTAL)", false, function(state)
    GodModeEnabled = state
    local char = LocalPlayer.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    
    if state then
        if hum then
            hum.MaxHealth = 5000
            hum.Health = 5000
        end
        setupGodMode(char)
        Window:Notify({Title = "GodMode", Content = "¡Modo Dios Activado! (Sin Limbo)", Type = "success"})
    else
        if hum then
            hum.MaxHealth = 100
            hum.Health = 100
        end
        if healthLoop then healthLoop:Disconnect() end
        Window:Notify({Title = "GodMode", Content = "Modo Dios Desactivado.", Type = "warning"})
    end
end)

-- 🔥 3. BOTÓN DE EMERGENCIA (Por si te vuelves fantasma)
TabMain:AddButton("🔄 FORZAR RESPAWN (Si eres fantasma)", function()
    local char = LocalPlayer.Character
    if char then
        local hum = char:FindFirstChildOfClass("Humanoid")
        if hum then
            -- Desactivamos el GodMode y forzamos la muerte para que el servidor nos reviva
            GodModeEnabled = false
            if healthLoop then healthLoop:Disconnect() end
            hum.Health = 0
        end
    end
    Window:Notify({Title = "Respawn", Content = "Forzando respawn...", Type = "info"})
end)
